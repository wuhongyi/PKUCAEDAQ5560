cmake_minimum_required(VERSION 3.16)
project(DecodeAndSortAll_R5560 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ROOT 6.20 REQUIRED COMPONENTS Core RIO Tree Hist)
add_library(r5560_decode R5560Decode.cc)
target_include_directories(r5560_decode PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(DecodeAndSortAll DecodeAndSortAll.cc)
target_link_libraries(DecodeAndSortAll PRIVATE r5560_decode ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist)
add_custom_command(TARGET DecodeAndSortAll POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
  $<TARGET_FILE:DecodeAndSortAll> $<TARGET_FILE_DIR:DecodeAndSortAll>/decodeandsort${CMAKE_EXECUTABLE_SUFFIX})
if(MSVC)
  target_compile_options(r5560_decode PRIVATE /W4 /utf-8)
  target_compile_options(DecodeAndSortAll PRIVATE /W4 /utf-8)
else()
  target_compile_options(r5560_decode PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(DecodeAndSortAll PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_decode tests/test_decode.cc)
  target_link_libraries(test_decode PRIVATE r5560_decode)
  add_test(NAME decode_and_sort COMMAND test_decode)
  add_executable(test_window_sort tests/test_window_sort.cc)
  target_link_libraries(test_window_sort PRIVATE r5560_decode)
  add_test(NAME rolling_window_sort COMMAND test_window_sort)
  add_executable(check_root tests/check_root.cc)
  target_include_directories(check_root PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(check_root PRIVATE ROOT::Core ROOT::RIO ROOT::Tree ROOT::Hist)
endif()
install(TARGETS DecodeAndSortAll RUNTIME DESTINATION bin)
